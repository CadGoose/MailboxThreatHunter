Incident checklist
# BEC containment checklist for cfo@contoso.example (review before running anything)

## 1. Contain
Revoke-MgUserSignInSession -UserId 'cfo@contoso.example'                     # kill refresh tokens / sessions
Update-MgUser -UserId 'cfo@contoso.example' -AccountEnabled:$false           # if takeover is confirmed
# Reset the password out-of-band and require MFA re-registration (Entra > user > Authentication methods)

## 2. Remove persistence
Get-InboxRule -Mailbox 'cfo@contoso.example' | Format-List Name,Description    # export first, then Remove-InboxRule the malicious ones
Set-Mailbox -Identity 'cfo@contoso.example' -ForwardingSmtpAddress $null -ForwardingAddress $null -DeliverToMailboxAndForward $false
Get-EXOMailboxPermission -Identity 'cfo@contoso.example' | Where-Object { -not $_.IsInherited }   # remove unexpected FullAccess
Get-MgUserAuthenticationMethod -UserId 'cfo@contoso.example'                 # delete authenticator/phone the user did not add
# Revoke unknown OAuth grants: Get-MgOauth2PermissionGrant -Filter "principalId eq '<user-object-id>'"

## 3. Scope
Search-UnifiedAuditLog -StartDate (Get-Date).AddDays(-30) -EndDate (Get-Date) -UserIds 'cfo@contoso.example' -ResultSize 5000
# Sign-in logs: filter user, look for new countries / IPs / legacy clients (see entra-signin-analyzer)
# Message trace: mail SENT by the user since first suspicious sign-in (invoice/payment-change requests to clients?)
# MailItemsAccessed (E5/Audit Premium) to determine what was read

## 4. Notify and recover

# Recall or follow up on any outbound fraudulent mail; contact the bank immediately for wires
